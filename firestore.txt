rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Koleksi Pengguna (users)
    match /users/{userId} {
      allow create: if request.auth != null && (request.auth.uid == userId || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      allow read, update: if request.auth != null && (request.auth.uid == userId || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      allow delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Koleksi Pesanan (orders)
    match /orders/{orderId} {
      allow read, write: if request.auth != null && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan' || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      allow delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Aturan untuk data pelanggan (customers)
    match /customers/{customerId} {
      allow read, write: if request.auth != null;
    }

    // Koleksi Log Galon Kembali (returned_gallons_log)
    match /returned_gallons_log/{logId} {
      // Karyawan bisa membuat dan membaca log.
      allow read, create: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan';
      // Admin bisa membaca dan menghapus log.
      allow read, delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Koleksi Penjualan Harian (daily_sales)
    match /daily_sales/{saleId} {
      // --- PERBAIKAN: Berikan izin tulis (write) kepada Admin ---
      // Karyawan dan Admin sekarang bisa membaca dan menulis data penjualan harian.
      allow read, write: if request.auth != null && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan' || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      // Hanya Admin yang bisa menghapus rekap penjualan.
      allow delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Koleksi Metadata Sinkronisasi Harian (daily_sync_metadata)
    match /daily_sync_metadata/{dateDocId} {
      allow read, delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
      allow write: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan';
    }

    // Koleksi Stok Galon Harian (daily_stock_levels)
    match /daily_stock_levels/{stockDateId} {
      allow read: if request.auth != null && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan' || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      allow create, update: if request.auth != null && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'karyawan' || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      allow delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Aturan default: tolak semua akses ke koleksi lain jika tidak ada aturan spesifik
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
